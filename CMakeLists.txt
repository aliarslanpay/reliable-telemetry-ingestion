cmake_minimum_required(VERSION 3.22)
project(reliable_telemetry_ingestion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TELEMETRY_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(TELEMETRY_TSAN "Enable ThreadSanitizer" OFF)
if(TELEMETRY_SANITIZE AND TELEMETRY_TSAN)
  message(FATAL_ERROR "Use separate ASan/UBSan and TSan build directories")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(TELEMETRY_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(TELEMETRY_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(telemetry_contract src/contract.cpp)
target_include_directories(telemetry_contract PUBLIC include)
target_link_libraries(telemetry_contract PUBLIC OpenSSL::Crypto nlohmann_json::nlohmann_json)
include(CTest)
if(BUILD_TESTING)
  add_executable(contract_test tests/contract_test.cpp)
  target_link_libraries(contract_test PRIVATE telemetry_contract)
  add_test(NAME contract COMMAND contract_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/events.json)
endif()
